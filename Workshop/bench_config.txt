# ============================================================
# -- Batch curve : m(n) for DUPLICATES of the same ingredient --
#    "Duplicates" = several copies of the SAME name/stage in the basket
#    (e.g. 4x Kanda leaf/leaf). Distinct ingredients are NOT affected:
#    they still sum normally (diversity concentrates). Stacking the same
#    thing does NOT multiply x N -- it follows the bell curve below.
#
#    HOW TO READ 'Batch': a flat list of [min,max] PAIRS, per-mille
#    (1000 = x1.0 = unchanged). Pair #1 is for n=1 copy, pair #2 for
#    n=2, ... one pair per count. 6 pairs here => n = 1..6.
#    At each cook, for a group of n copies the engine draws ONE random
#    value inside that pair's [min,max] and multiplies the group's axis
#    contribution by it. The min..max spread IS the natural fuzz: two
#    identical batches are never exactly the same.
#
#    SHAPE (bell): rises to a sweet spot (~n=3-4, peak ~x1.5), then
#    falls back (too many copies = diminishing, then worse). Edit the
#    numbers to reshape; keep an EVEN count of values (min,max pairs).
#    Beyond the last pair the last one is reused (but NMax caps first).
#
#    OVER-SATURATION: n >= NMax => batch SPOILED. Instead of the curve,
#    potency collapses: a draw in [SpoilMin..SpoilMax] per-mille is
#    applied to the output effect (sibling of 'overcooked'). Matter is
#    still consumed. "Too much kills the mix."
#
#    (The per-axis natural variation is a SEPARATE knob: FluxAbs above.)
# ------------------------------------------------------------
NMax = 7                 # n >= NMax -> batch spoiled (potency collapses)
SpoilMin = 0             # spoil draw, per-mille: effect multiplier in [SpoilMin..SpoilMax]
SpoilMax = 150           #   e.g. 0..150 = output effect falls to 0%..15%
# m(n) table, per-mille, [min,max] pairs for n=1..6.  NO inline # on this line.
#        n=1        n=2        n=3(peak)  n=4(peak)  n=5        n=6
Batch = 950,1050, 1200,1300, 1400,1500, 1350,1500, 950,1150, 550,800
# ============================================================
